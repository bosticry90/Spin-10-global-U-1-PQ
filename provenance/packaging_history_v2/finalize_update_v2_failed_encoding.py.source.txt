from pathlib import Path
import json,hashlib,zipfile,subprocess
from datetime import datetime,timezone
H=Path(__file__).resolve().parent
R=H/'Spin-10-global-U-1-PQ'
def sha(b): return hashlib.sha256(b).hexdigest()
def dump(p,v): p.write_text(json.dumps(v,indent=2)+'\n',encoding='utf-8',newline='\n')
source_path=R/'provenance/source_manifest_v2.json'
m=json.loads(source_path.read_text())
auth=json.loads((H/'claim_authentication_v2.json').read_text())
assert auth['execution_state']=='COMPLETED'
filename='spin10_pq_source_update_v2_history.zip'
if not any(a['filename']==filename for a in m['archives']):
    existing={e['path']:e for e in m['historical_files']}
    added=[]
    for check in auth['checks']:
        for historical in check['historical_policy_inputs']:
            path='historical/'+historical['authenticated_commit']+'/'+historical['path']
            data=Path(historical['export_file']).read_bytes()
            assert sha(data)==historical['expected_sha256']
            if path in existing:
                assert existing[path]['sha256']==sha(data)
                continue
            added.append((path,data,historical))
    with zipfile.ZipFile(R/'reproducibility'/filename,'w',compression=zipfile.ZIP_DEFLATED) as z:
        for path,data,historical in added:
            z.writestr(path,data)
            m['historical_files'].append({'path':path,'sha256':sha(data),'size_bytes':len(data),'historical_commit':historical['authenticated_commit'],'storage':'UPDATE_V2','archive':filename})
    data=(R/'reproducibility'/filename).read_bytes()
    m['archives'].append({'filename':filename,'sha256':sha(data),'size_bytes':len(data),'file_count':len(added)})
    dump(source_path,m)

def release_manifest():
    paths=subprocess.check_output(['git','-C',str(R),'ls-files','--cached','--others','--exclude-standard','-z']).decode().split('\0')
    paths=sorted(set(p for p in paths if p and p!='provenance/publication_manifest_v2.json'))
    files=[]
    for path in paths:
        p=R/path
        if p.is_file():
            data=p.read_bytes(); files.append({'path':path,'sha256':sha(data),'size_bytes':len(data)})
    dump(R/'provenance/publication_manifest_v2.json',{'schema':'SPIN10_PQ_PUBLICATION_MANIFEST_V2','source_commit':m['source_commit'],'manifest_self_binding':'The Git commit binds this manifest. It does not hash itself.','release_files':files,'scientific_acceptance':'NOT_PERFORMED'})
